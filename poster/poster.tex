\block{Contexte \& Probl\'ematique}{%
  Le \textbf{reverse engineering} consiste \`a comprendre un logiciel
  \textit{sans acc\`es au code source}. Ghidra, l'outil de r\'ef\'erence
  de la NSA, g\'en\`ere du code d\'ecompil\'e illisible :

  \vspace{0.5em}
  \begin{center}
  \begin{tikzpicture}
    \node[rectangle, rounded corners=5pt, fill=LOrange,
          draw=Orange, line width=2pt,
          text width=28cm, inner sep=12pt]{%
      \large
      \textbf{\color{Orange}Probl\`eme :}\quad
      \texttt{FUN\_00401000(param\_1, param\_2)} ---
      impossible \`a comprendre sans analyse manuelle
      longue et sujette aux erreurs.
    };
  \end{tikzpicture}
  \end{center}

  \vspace{0.5em}
  \begin{tabular}{@{}ll}
    \textcolor{Green}{$\checkmark$} & Renommer fonctions et param\`etres \\[0.4em]
    \textcolor{Green}{$\checkmark$} & D\'etecter les APIs utilis\'ees \\[0.4em]
    \textcolor{Green}{$\checkmark$} & G\'en\'erer des commentaires explicatifs \\[0.4em]
    \textcolor{Green}{$\checkmark$} & Comparer trois LLMs locaux sur ces t\^aches \\
  \end{tabular}
}

\block{Approche Multi-Pass It\'erative}{%
  Chaque passe \textbf{enrichit le JSON} avec les r\'esultats
  de la passe pr\'ec\'edente. Le LLM dispose ainsi d'un contexte
  progressivement plus riche.

  \vspace{0.6em}
  \begin{center}
  \renewcommand{\arraystretch}{1.6}
  \begin{tabular}{@{}clp{18cm}@{}}
    \textbf{\color{Blue}P1}
      & \textbf{Renommage}
      & Propose des noms descriptifs pour chaque fonction \\
    \multicolumn{3}{c}{{\small\color{DGray}$\downarrow$ enrichissement du JSON extrait}} \\
    \textbf{\color{Green}P2}
      & \textbf{Analyse}
      & Types de retour, APIs, signatures (avec noms de P1) \\
    \multicolumn{3}{c}{{\small\color{DGray}$\downarrow$ enrichissement du JSON extrait}} \\
    \textbf{\color{Orange}P3}
      & \textbf{Commentaires}
      & Commentaires Doxygen avec contexte maximal \\
  \end{tabular}
  \end{center}

  \vspace{0.4em}
  \begin{center}
  \begin{tikzpicture}
    \node[rectangle, rounded corners=5pt, fill=LGreen,
          draw=Green, line width=2pt,
          inner sep=10pt, text width=28cm, align=center]{%
      \texttt{func\_001}
      $\;\xrightarrow{\;\mathbf{P1}\;}$
      \texttt{copy\_alloc}
      $\;\xrightarrow{\;\mathbf{P2}\;}$
      \texttt{void*(void* src, size\_t n)}
      $\;\xrightarrow{\;\mathbf{P3}\;}$
      commentaire Doxygen
    };
  \end{tikzpicture}
  \end{center}
}

\block{M\'etriques d'\'evaluation}{%
  \renewcommand{\arraystretch}{1.5}
  \begin{tabular}{@{}lp{22cm}@{}}
    \toprule
    \textbf{M\'etrique} & \textbf{D\'efinition} \\
    \midrule
    \textbf{Semantic Score}
      & Similarit\'e cosinus entre nom sugg\'er\'e et nom attendu \\
    \textbf{Exact Match}
      & Pourcentage de noms identiques au ground truth \\
    \textbf{Hallucination Rate}
      & Proportion de suggestions totalement incorrectes \\
    \textbf{BLEU / ROUGE-L}
      & Qualit\'e des commentaires g\'en\'er\'es (passe\,3) \\
    \bottomrule
  \end{tabular}
}
